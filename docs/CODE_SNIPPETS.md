## FeedViewModel and UiState

```kotlin
data class FeedUiState(
    val isLoading: Boolean = false,
    val items: List<FeedItem> = emptyList(),
    val errorMessage: String? = null,
)

sealed interface FeedEffect {
    data class OpenDetail(val id: String) : FeedEffect
    data object ShowRetrySnackbar : FeedEffect
}

@HiltViewModel
class FeedViewModel @Inject constructor(
    private val repository: FeedRepository,
    private val savedStateHandle: SavedStateHandle,
) : ViewModel() {

    private val _uiState = MutableStateFlow(FeedUiState(isLoading = true))
    val uiState: StateFlow<FeedUiState> = _uiState.asStateFlow()

    private val _effects = Channel<FeedEffect>(Channel.BUFFERED)
    val effects: Flow<FeedEffect> = _effects.receiveAsFlow()

    val query: StateFlow<String> = savedStateHandle.getStateFlow("query", "")

    init {
        viewModelScope.launch {
            repository.observeFeed()
                .catch { e -> _uiState.update { it.copy(isLoading = false, errorMessage = e.toUserMessage()) } }
                .collect { items -> _uiState.update { it.copy(items = items, isLoading = false) } }
        }
    }

    fun onItemClick(id: String) = viewModelScope.launch { _effects.send(FeedEffect.OpenDetail(id)) }
}
```

## Navigation 3 with NavDisplay

```kotlin
@Serializable data object Home : NavKey
@Serializable data class Detail(val id: String) : NavKey

@Composable
fun AppNavDisplay() {
    val backStack = rememberNavBackStack(Home)
    NavDisplay(
        backStack = backStack,
        onBack = { backStack.removeLastOrNull() },
        entryProvider = entryProvider {
            entry<Home> {
                HomeRoute(onOpen = { id -> backStack.add(Detail(id)) })
            }
            entry<Detail> { key ->
                DetailRoute(id = key.id, onBack = { backStack.removeLastOrNull() })
            }
        },
    )
}
```

## OfflineFirstFeedRepository

```kotlin
interface FeedRepository {
    fun observeFeed(): Flow<List<FeedItem>>            // reads Room, always
    suspend fun refresh(): Result<Unit>                 // network -> Room
}

class OfflineFirstFeedRepository @Inject constructor(
    private val dao: FeedDao,
    private val api: FeedApi,
    private val @IoDispatcher io: CoroutineDispatcher,
) : FeedRepository {

    override fun observeFeed(): Flow<List<FeedItem>> =
        dao.observeAll().map { rows -> rows.map(FeedEntity::toModel) }

    override suspend fun refresh(): Result<Unit> = withContext(io) {
        runCatching {
            val remote = api.getFeed()
            dao.upsertAll(remote.map(FeedDto::toEntity))
        }
    }
}
```

## GitHub Actions pipeline

```yaml
name: Android CI

on:
  push:
    branches: [main]
    tags: ['v*']
  pull_request:
    branches: ['*']

concurrency:
  group: ci-${{ github.ref }}
  cancel-in-progress: true

permissions:
  contents: write

jobs:
  verify:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-java@v4
        with:
          distribution: temurin
          java-version: '21'
      - uses: gradle/actions/setup-gradle@v4
      - name: Unit tests and lint
        run: ./gradlew testDebugUnitTest lintDebug --stacktrace
      - name: Assemble debug
        run: ./gradlew assembleDebug
      - uses: actions/upload-artifact@v4
        with:
          name: app-debug
          path: app/build/outputs/apk/debug/*.apk

  release:
    if: startsWith(github.ref, 'refs/tags/v')
    needs: verify
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-java@v4
        with:
          distribution: temurin
          java-version: '21'
      - uses: gradle/actions/setup-gradle@v4
      - name: Decode upload keystore
        run: echo "${{ secrets.KEYSTORE_BASE64 }}" | base64 -d > $RUNNER_TEMP/release.jks
      - name: Build signed AAB and APK
        env:
          KEYSTORE_PATH: ${{ runner.temp }}/release.jks
          KEYSTORE_PASSWORD: ${{ secrets.KEYSTORE_PASSWORD }}
          KEY_ALIAS: ${{ secrets.KEY_ALIAS }}
          KEY_PASSWORD: ${{ secrets.KEY_PASSWORD }}
        run: ./gradlew bundleRelease assembleRelease --stacktrace
      - uses: actions/upload-artifact@v4
        with:
          name: mapping
          path: app/build/outputs/mapping/release/mapping.txt
      - uses: softprops/action-gh-release@v2
        with:
          files: |
            app/build/outputs/apk/release/*.apk
            app/build/outputs/bundle/release/*.aab
```

## AGENTS.md single-module template

```markdown
# [App name]

Native Android app in Kotlin. One module, Single Activity, Compose, Material 3.

## Stack
- AGP 9.3.2, Gradle 9.7.1, Kotlin 2.3, KSP >= 2.3.6, Hilt >= 2.59.2
- compileSdk 37 (fall back to 36 if a library lacks API 37 metadata), minSdk 24
- Room, DataStore, Retrofit + OkHttp, kotlinx.serialization
- Navigation 3 (NavDisplay + @Serializable NavKey), Coroutines/Flow

## Packages - com.<you>.<app>
- ui/theme/{Color.kt, Type.kt, Theme.kt}
- ui/navigation/{NavigationKeys.kt, Navigation.kt}
- ui/screens/<feature>/{<Feature>Screen.kt, ViewModel.kt, UiState.kt}
- data/local/ (Room), data/remote/ (Retrofit), data/repository/
- domain/model/ (pure Kotlin); add domain/usecase/ only when reused
- di/ per layer (DataModule.kt, DatabaseModule.kt)

## Rules
- One StateFlow<UiState> per screen, stateIn(WhileSubscribed(5_000))
- One-shot effects in a Channel; never inside the state object
- SavedStateHandle for user input (rotation survives, process death does not)
- Room is the source of truth; the network writes into Room
- Inject dispatchers; never reference Dispatchers.IO inline
- No business logic in composables; hoist state
- enableEdgeToEdge() before setContent; consume Scaffold innerPadding
- No feature package imports another feature package
- Do not add Gradle modules until there are 4+ features

## Commands
- Build: ./gradlew assembleDebug
- Test: ./gradlew testDebugUnitTest
- Lint: ./gradlew lintDebug
- Single test: ./gradlew testDebugUnitTest --tests "*FeedViewModelTest"

## Do not
- Run ./gradlew clean unless the cache is corrupt
- Call bare gradle; always the wrapper
- Commit keystores, local.properties or secrets
- Add -keep rules before running :app:analyzeReleaseR8Config
```
